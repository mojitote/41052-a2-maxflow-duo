cmake_minimum_required(VERSION 3.16)
project(maxflow_two_person LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_DINIC "Build the two-algorithm comparison" OFF)
option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(maxflow src/maxflow.cpp)
target_include_directories(maxflow PUBLIC include)
if(ENABLE_DINIC)
  target_compile_definitions(maxflow PUBLIC MF_ENABLE_DINIC)
endif()
foreach(program benchmark benchmark_cpu)
  add_executable(${program} src/${program}.cpp)
  target_link_libraries(${program} PRIVATE maxflow)
endforeach()
add_executable(test_maxflow tests/test_maxflow.cpp)
target_link_libraries(test_maxflow PRIVATE maxflow)
enable_testing()
add_test(NAME certificates COMMAND test_maxflow)
